包装器骨架。⚠️ `const w = (fromId, toId?) => beatWindow(...)` 这一行的**字面
形态是解析器契约**：check_script.py 的 SCENE_CALL_RE 只认 `w('id','id')`，
把它改名或抽成 hook 会让 `--check-scenes` 的分镜↔代码互比静默失效。

  export const P0Name: React.FC<{scene: SceneRange}> = ({scene}) => {
    const w = (fromId: string, toId?: string) => beatWindow(scene.sentences, scene.from, fromId, toId);
    // at() = 时点锚（只取某句起始帧，不是分镜窗口）。刻意不叫 w()——混用会让
    // 时点锚刷出「分镜陈旧」假 WARN。
    const at = (id: string) => w(id).from;
    const bA = w('p0-01', 'p0-03');
    return (
      <AbsoluteFill>
        <Sequence {...bA} name="0-A 分镜标题（与 storyboard.md 镜号一致）">
          <YourOneOffComponent freezeAt={at('p0-03') - bA.from} />
        </Sequence>
      </AbsoluteFill>
    );
  };
